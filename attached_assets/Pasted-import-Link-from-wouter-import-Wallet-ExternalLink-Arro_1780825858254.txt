import { Link } from "wouter";
import { Wallet, ExternalLink, ArrowRightLeft, TrendingUp, TrendingDown, RefreshCw } from "lucide-react";
import { useQuery } from "@tanstack/react-query";

interface CoinMarket {
  id: string; symbol: string; name: string; image: string;
  current_price: number; price_change_percentage_24h: number;
  market_cap: number; total_volume: number;
}

const COIN_ICONS: Record<string, string> = {
  bitcoin: "₿", ethereum: "Ξ", binancecoin: "B", solana: "◎", ripple: "✕",
  cardano: "₳", dogecoin: "Ð", "avalanche-2": "A", "matic-network": "⬡", litecoin: "Ł",
};

function formatMarketCap(n: number) {
  if (n >= 1e12) return `$${(n / 1e12).toFixed(2)}T`;
  if (n >= 1e9)  return `$${(n / 1e9).toFixed(2)}B`;
  if (n >= 1e6)  return `$${(n / 1e6).toFixed(2)}M`;
  return `$${n.toLocaleString()}`;
}

async function fetchMarkets(): Promise<CoinMarket[]> {
  const res = await fetch(
    "https://api.coingecko.com/api/v3/coins/markets?vs_currency=usd&order=market_cap_desc&per_page=8&page=1&sparkline=false&price_change_percentage=24h"
  );
  if (!res.ok) throw new Error("Failed to fetch market data");
  return res.json();
}

export default function Overview() {
  const { data: markets, isLoading, isError, refetch, dataUpdatedAt } = useQuery<CoinMarket[]>({
    queryKey: ["coin-markets"],
    queryFn: fetchMarkets,
    refetchInterval: 30_000,
    staleTime: 15_000,
  });
  const lastUpdated = dataUpdatedAt ? new Date(dataUpdatedAt).toLocaleTimeString() : null;

  return (
    <div className="max-w-6xl mx-auto space-y-8">
      <header className="flex flex-col md:flex-row md:items-end justify-between gap-4">
        <div>
          <h1 className="text-3xl font-display font-bold">Overview</h1>
          <p className="text-muted-foreground mt-1">Welcome back, Test User</p>
        </div>
        <div className="flex gap-3">
          <Link href="/deposit">
            <button className="bg-secondary text-secondary-foreground hover:bg-secondary/80 h-12 px-6 rounded-lg font-semibold transition-all active:scale-95">Deposit</button>
          </Link>
          <Link href="/trade">
            <button className="bg-primary text-primary-foreground shadow-lg shadow-primary/20 hover:bg-primary/90 h-12 px-6 rounded-lg font-semibold transition-all active:scale-95">Trade</button>
          </Link>
        </div>
      </header>

      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
        <div className="rounded-2xl border border-border p-6 md:col-span-2 bg-gradient-to-br from-secondary to-card relative overflow-hidden shadow-xl shadow-black/40">
          <div className="absolute top-0 right-0 w-64 h-64 bg-primary/5 rounded-full blur-3xl -translate-y-1/2 translate-x-1/3 pointer-events-none" />
          <div className="flex items-center gap-2 text-muted-foreground font-medium mb-2">
            <Wallet className="w-4 h-4" /><span>Estimated Balance</span>
          </div>
          <div className="text-4xl md:text-5xl font-display font-bold mono-nums mb-2">$10,000.00</div>
          <div className="text-sm text-muted-foreground">
            Available USD: <span className="text-foreground font-medium mono-nums">$10,000.00</span>
          </div>
        </div>

        <div className="rounded-2xl border border-border bg-card p-6 flex flex-col justify-center shadow-xl shadow-black/40">
          <h3 className="text-muted-foreground font-medium mb-4">Quick Links</h3>
          <div className="grid grid-cols-2 gap-3">
            <Link href="/portfolio">
              <button className="flex items-center justify-between px-4 py-3 rounded-lg border border-border hover:bg-secondary transition-all group w-full">
                <span className="text-sm font-medium">Portfolio</span>
                <ExternalLink className="w-4 h-4 text-muted-foreground group-hover:text-foreground" />
              </button>
            </Link>
            <Link href="/transactions">
              <button className="flex items-center justify-between px-4 py-3 rounded-lg border border-border hover:bg-secondary transition-all group w-full">
                <span className="text-sm font-medium">History</span>
                <ExternalLink className="w-4 h-4 text-muted-foreground group-hover:text-foreground" />
              </button>
            </Link>
            <Link href="/trade" className="col-span-2">
              <button className="flex items-center justify-center gap-2 px-4 py-3 rounded-lg bg-secondary/50 hover:bg-secondary transition-all w-full">
                <ArrowRightLeft className="w-4 h-4" />
                <span className="text-sm font-medium">Go to Markets</span>
              </button>
            </Link>
          </div>
        </div>
      </div>

      <div className="space-y-4">
        <div className="flex items-center justify-between">
          <div>
            <h2 className="text-xl font-display font-bold">Live Crypto Markets</h2>
            {lastUpdated && <p className="text-xs text-muted-foreground mt-0.5">Updated {lastUpdated} · auto-refreshes every 30s</p>}
          </div>
          <div className="flex items-center gap-3">
            <button onClick={() => refetch()} className="flex items-center gap-1.5 text-xs text-muted-foreground hover:text-foreground transition-colors">
              <RefreshCw className="w-3.5 h-3.5" /> Refresh
            </button>
            <Link href="/trade" className="text-sm font-medium text-primary hover:underline">View All</Link>
          </div>
        </div>

        <div className="rounded-2xl border border-border bg-card overflow-hidden shadow-xl shadow-black/40">
          <div className="overflow-x-auto">
            <table className="w-full text-sm text-left">
              <thead className="text-xs text-muted-foreground uppercase bg-secondary/30">
                <tr>
                  <th className="px-6 py-4 font-medium">Name</th>
                  <th className="px-6 py-4 font-medium text-right">Price</th>
                  <th className="px-6 py-4 font-medium text-right">24h Change</th>
                  <th className="px-6 py-4 font-medium text-right hidden md:table-cell">Market Cap</th>
                  <th className="px-6 py-4 font-medium text-right hidden lg:table-cell">Volume 24h</th>
                  <th className="px-6 py-4 font-medium text-center">Action</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-border">
                {isLoading
                  ? Array.from({ length: 6 }).map((_, i) => (
                      <tr key={i} className="animate-pulse">
                        <td className="px-6 py-4"><div className="flex items-center gap-3"><div className="w-8 h-8 rounded-full bg-secondary/60" /><div className="h-4 bg-secondary/60 rounded w-24" /></div></td>
                        <td className="px-6 py-4 text-right"><div className="h-4 bg-secondary/60 rounded w-20 ml-auto" /></td>
                        <td className="px-6 py-4 text-right"><div className="h-4 bg-secondary/60 rounded w-14 ml-auto" /></td>
                        <td className="px-6 py-4 text-right hidden md:table-cell"><div className="h-4 bg-secondary/60 rounded w-16 ml-auto" /></td>
                        <td className="px-6 py-4 hidden lg:table-cell"><div className="h-4 bg-secondary/60 rounded w-16 ml-auto" /></td>
                        <td className="px-6 py-4"><div className="h-8 bg-secondary/60 rounded w-14 mx-auto" /></td>
                      </tr>
                    ))
                  : isError
                  ? <tr><td colSpan={6} className="px-6 py-10 text-center text-muted-foreground text-sm">Unable to load live market data. <button onClick={() => refetch()} className="text-primary hover:underline">Try again</button></td></tr>
                  : markets?.map((coin) => {
                      const up   = coin.price_change_percentage_24h >= 0;
                      const icon = COIN_ICONS[coin.id];
                      return (
                        <tr key={coin.id} className="hover:bg-secondary/30 transition-colors">
                          <td className="px-6 py-4">
                            <div className="flex items-center gap-3">
                              {icon
                                ? <span className="text-2xl w-8 text-center">{icon}</span>
                                : <img src={coin.image} alt={coin.name} className="w-8 h-8 rounded-full" />}
                              <div>
                                <div className="font-bold text-foreground">{coin.symbol.toUpperCase()}</div>
                                <div className="text-xs text-muted-foreground">{coin.name}</div>
                              </div>
                            </div>
                          </td>
                          <td className="px-6 py-4 text-right mono-nums font-medium text-foreground">
                            ${coin.current_price.toLocaleString(undefined, { minimumFractionDigits: coin.current_price < 1 ? 4 : 2, maximumFractionDigits: coin.current_price < 1 ? 6 : 2 })}
                          </td>
                          <td className="px-6 py-4 text-right mono-nums font-medium">
                            <div className={`flex items-center justify-end gap-1 ${up ? "text-success" : "text-destructive"}`}>
                              {up ? <TrendingUp className="w-3 h-3" /> : <TrendingDown className="w-3 h-3" />}
                              {up ? "+" : ""}{coin.price_change_percentage_24h.toFixed(2)}%
                            </div>
                          </td>
                          <td className="px-6 py-4 text-right mono-nums text-muted-foreground hidden md:table-cell">{formatMarketCap(coin.market_cap)}</td>
                          <td className="px-6 py-4 text-right mono-nums text-muted-foreground text-xs hidden lg:table-cell">{formatMarketCap(coin.total_volume)}</td>
                          <td className="px-6 py-4 text-center">
                            <Link href="/trade">
                              <button className="bg-primary/10 text-primary hover:bg-primary/20 border border-primary/20 h-8 px-4 rounded-md text-xs font-semibold transition-colors">
                                Trade
                              </button>
                            </Link>
                          </td>
                        </tr>
                      );
                    })}
              </tbody>
            </table>
          </div>
        </div>
      </div>
    </div>
  );
}

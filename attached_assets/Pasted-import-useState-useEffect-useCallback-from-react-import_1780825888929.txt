interface ForexRate {
  pair: string; base: string; quote: string;
  rate: number; prevRate: number; change: number; changePct: number; spread: number;
}
interface CryptoMarket {
  id: string; symbol: string; name: string; image: string;
  current_price: number; price_change_percentage_1h_in_currency: number;
  price_change_percentage_24h: number;
}
interface Position {
  id: number; market: "forex" | "crypto"; pair: string;
  side: "buy" | "sell"; size: number; leverage: number;
  openPrice: number; sl: number; tp: number; openedAt: string;
}

const PAIR_CFG = [
  { pair:"EUR/USD", invert:true,  k1:"EUR", k2:"",    digits:5, spread:0.00012 },
  { pair:"GBP/USD", invert:true,  k1:"GBP", k2:"",    digits:5, spread:0.00015 },
  { pair:"USD/JPY", invert:false, k1:"",    k2:"JPY",  digits:3, spread:0.012   },
  { pair:"USD/CHF", invert:false, k1:"",    k2:"CHF",  digits:5, spread:0.00014 },
  { pair:"AUD/USD", invert:true,  k1:"AUD", k2:"",    digits:5, spread:0.00013 },
  { pair:"USD/CAD", invert:false, k1:"",    k2:"CAD",  digits:5, spread:0.00014 },
  { pair:"NZD/USD", invert:true,  k1:"NZD", k2:"",    digits:5, spread:0.00016 },
  { pair:"EUR/GBP", invert:false, k1:"EUR", k2:"GBP",  digits:5, spread:0.00013 },
  { pair:"EUR/JPY", invert:false, k1:"EUR", k2:"JPY",  digits:3, spread:0.014   },
  { pair:"GBP/JPY", invert:false, k1:"GBP", k2:"JPY",  digits:3, spread:0.016   },
  { pair:"USD/SGD", invert:false, k1:"",    k2:"SGD",  digits:5, spread:0.0002  },
  { pair:"USD/HKD", invert:false, k1:"",    k2:"HKD",  digits:4, spread:0.0005  },
];
const LOT_SIZES  = [0.01, 0.05, 0.1, 0.25, 0.5, 1.0];
const LEVERAGES  = [1, 10, 50, 100, 200];
const CRYPTO_LVG = [1, 2, 5, 10, 25];

function computeForex(rates: Record<string,number>, cfg: typeof PAIR_CFG[0]) {
  const { invert, k1, k2 } = cfg;
  if (invert && k1) return 1 / (rates[k1] ?? 1);
  if (!invert && k2 && !k1) return rates[k2] ?? 1;
  if (k1 && k2) return (rates[k2] ?? 1) / (rates[k1] ?? 1);
  return 1;
}

async function fetchForex(): Promise<Record<string,number>> {
  const r = await fetch("https://cdn.jsdelivr.net/npm/@fawazahmed0/currency-api@latest/v1/currencies/usd.json");
  if (!r.ok) throw new Error("forex");
  const d = await r.json();
  return Object.fromEntries(Object.entries(d.usd as Record<string,number>).map(([k,v])=>[k.toUpperCase(),v]));
}
async function fetchCrypto(): Promise<CryptoMarket[]> {
  const ids = "bitcoin,ethereum,solana,binancecoin,ripple,cardano,dogecoin,avalanche-2,polkadot,chainlink,litecoin,uniswap";
  const r = await fetch(`https://api.coingecko.com/api/v3/coins/markets?vs_currency=usd&ids=${ids}&order=market_cap_desc&per_page=12&page=1&sparkline=false&price_change_percentage=1h,24h`);
  if (!r.ok) throw new Error("crypto");
  return r.json();
}
